\section{Research Experience}

\cventry{2026--present}{Excitons and phonons in strongly correlated materials}{UT Austin}{}{}{%
\textit{With Antonios Alvertis.} Computing exciton binding energies in strongly coupled electron-phonon systems
with tensor-network methods, in the regime where perturbative treatments of the
coupling break down.}

\cventry{2025--2026}{Generalized virial identities}{UT Austin}{}{\cvtag{Independent project}}{%
Derived a one-parameter family of virial identities for $O(n)$-symmetric
solitons, instantons and bounces that resolves Derrick's global constraint into
core and tail contributions. Verified them analytically for the Fubini-Lipatov
instanton, the BPS monopole and the BPST instanton, and established their
residuals as an accuracy diagnostic for numerical solutions.}

\cventry{2024--2025}{Charm production across color environments}{ALICE Collaboration, UT Austin}{}{}{%
\textit{With Christina Markert.} Measured $h$--$D^0$ azimuthal correlations in ALICE Run-3 Pb--Pb data to probe
charm production in different color environments, and contributed performance
studies of the upgraded Inner Tracking System (ITS2) as collaboration service work.}

\cventry{2024}{High-redshift galaxies and top-heavy initial mass functions}{UT Austin}{}{}{%
\textit{With Katherine Freese.} Modeled high-redshift galaxy spectral energy distributions with the population
synthesis code P\'EGASE to determine where JWST observations remain consistent
with $\Lambda$CDM. \cvtag{Under review}}

\cventry{2020--2022}{Solitons in scalar field theories with non-degenerate vacua}{UNAM}{}{}{%
\textit{With Manuel Torres Labansat.} Established the existence of static $n$-kink structures in deformed potentials
by asymptotic analysis, characterized the inter-kink binding forces
numerically, and derived an analytical relation for the multi-kink energy.
Renormalized the one-loop quantum mass correction of kinks in a generalized
$\phi^4$ potential. \cvtag{Two refereed papers}}
